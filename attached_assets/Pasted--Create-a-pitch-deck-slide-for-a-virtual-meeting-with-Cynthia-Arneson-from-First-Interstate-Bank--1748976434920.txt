# If using Python with matplotlib:
import matplotlib.pyplot as plt

# Data for visualization
categories = ['Lack Financial Knowledge', 'Face Financial Anxiety', 'Potential Reach']
values = [26, 70, 100]  # 100 for potential reach as a placeholder for 1 million

plt.figure(figsize=(10, 6))
plt.bar(categories, values, color=['#1E90FF', '#3CB371', '#ADD8E6'])
plt.title('Impact of Financial Literacy Initiative', fontsize=16, pad=15)
plt.ylabel('Percentage or Reach (in millions)', fontsize=12)
plt.ylim(0, 120)
for i, v in enumerate(values):
    plt.text(i, v + 2, str(v) if i < 2 else '1M', ha='center', fontsize=10)
plt.savefig('pitch_slide.png')
print("Slide image saved as 'pitch_slide.png'")
